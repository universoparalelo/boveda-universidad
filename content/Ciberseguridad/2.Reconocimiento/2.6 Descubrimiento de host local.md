### Con un script propio para ICMP
```bash
#!/bin/bash

function ctrl_c(){
	echo -e "\n\n[!] Saliendo..."
	tput cnorm; exit 1
}

tput civis

trap ctrl_c

for i in $(seq 1 254); do
	for port in $(22,80,139,443,445,8080); do
		timeout 1 bash -o "echo '' >  /dev/tcp/192.168.111.$i/$port" && 2>/dev/null && echo -e [+]Host: 192.168.111.$i - Port: $port - [OPEN] &
	done
done

wait

tput cnorm
```
